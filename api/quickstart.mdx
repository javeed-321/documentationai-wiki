## Send an SMS

Use the following code to send a test message from your Twilio number.

<CodeGroup tabs="Node.js,Python">
```javascript
const twilio = require("twilio");
const client = twilio("YOUR_ACCOUNT_SID", "YOUR_AUTH_TOKEN");

client.messages
  .create({
    body: "Hello from Twilio",
    from: "+15005550006",
    to: "+15005550006",
  })
  .then((message) => console.log(message.sid));
```

```python
from twilio.rest import Client

client = Client("YOUR_ACCOUNT_SID", "YOUR_AUTH_TOKEN")

message = client.messages.create(
    body="Hello from Twilio",
    from_="+15005550006",
    to="+15005550006",
)
print(message.sid)
```
</CodeGroup>

## Make an outbound call

Use the following code to place a test call.

<CodeGroup tabs="Node.js,Python">
```javascript
const twilio = require("twilio");
const client = twilio("YOUR_ACCOUNT_SID", "YOUR_AUTH_TOKEN");

client.calls
  .create({
    url: "https://example.com/voice.xml",
    from: "+15005550006",
    to: "+15005550006",
  })
  .then((call) => console.log(call.sid));
```

```python
from twilio.rest import Client

client = Client("YOUR_ACCOUNT_SID", "YOUR_AUTH_TOKEN")

call = client.calls.create(
    url="https://example.com/voice.xml",
    from_="+15005550006",
    to="+15005550006",
)
print(call.sid)
```
</CodeGroup>
